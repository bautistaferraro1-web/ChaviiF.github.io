const BLOCKS = {
    grass:{c:"#62b53e",solid:true},
    dirt:{c:"#8b5a2b",solid:true},
    stone:{c:"#888888",solid:true},
    wood:{c:"#925c2b",solid:true},
    leaves:{c:"#3d9638",solid:true,transparent:true},
    water:{c:"#3f91dc",solid:false,transparent:true},
    base:{c:"#555555",solid:true}
};

const selectable = [
    "grass",
    "dirt",
    "stone",
    "wood",
    "leaves",
    "water"
];

let selected = 0;

const SIZE = 140;
const HEIGHT = 30;
const BASE = 2;
const VIEW = 20;

const world = new Map();

function key(x,y,z){
    return x + "," + y + "," + z;
}

function inWorld(x,y,z){
    return (
        x >= 0 &&
        x < SIZE &&
        y >= 0 &&
        y < HEIGHT &&
        z >= 0 &&
        z < SIZE
    );
}

function setB(x,y,z,type){
    if(inWorld(x,y,z)){
        world.set(key(x,y,z),type);
    }
}

function getB(x,y,z){
    if(!inWorld(x,y,z)){
        return null;
    }

    return world.get(key(x,y,z)) || null;
}

function delB(x,y,z){
    if(y >= BASE){
        world.delete(key(x,y,z));
    }
}

function solid(x,y,z){
    const type = getB(x,y,z);

    return !!(
        type &&
        BLOCKS[type] &&
        BLOCKS[type].solid
    );
}

function noise(x,z){
    return (
        Math.sin(x * 0.073) *
        Math.cos(z * 0.067) +
        Math.sin((x + z) * 0.021) * 0.6 +
        Math.cos((x - z) * 0.034) * 0.4
    );
}

function mountainNoise(x,z){
    return (
        Math.sin(x * 0.038) +
        Math.cos(z * 0.047) +
        Math.sin((x + z) * 0.018)
    ) / 3;
}

function heightAt(x,z){

    let h = Math.floor(
        5 +
        noise(x,z) * 3 +
        mountainNoise(x,z) * 7
    );

    return Math.max(
        3,
        Math.min(HEIGHT - 7,h)
    );
}

function lake(x,z){
    return (
        Math.sin(x * 0.055) +
        Math.cos(z * 0.047) +
        Math.sin((x + z) * 0.022)
    ) > 2.1;
}

function cave(x,y,z){

    if(y <= BASE + 1) return false;
    if(y > 21) return false;

    return (
        Math.sin(x * 0.23 + y * 0.13) *
        Math.cos(z * 0.19 - y * 0.11) +
        Math.sin(z * 0.31 + x * 0.09 + y * 0.07)
    ) > 1.25;
}

function generate(){

    world.clear();

    for(let x=0;x<SIZE;x++){
        for(let z=0;z<SIZE;z++){
            setB(x,0,z,"base");
            setB(x,1,z,"base");
        }
    }

    for(let x=0;x<SIZE;x++){

        for(let z=0;z<SIZE;z++){

            let h = lake(x,z)
                ? 4
                : heightAt(x,z);

            for(let y=BASE;y<=h;y++){

                if(cave(x,y,z)){
                    continue;
                }

                if(y === h){
                    setB(x,y,z,"grass");
                }
                else if(y >= h - 2){
                    setB(x,y,z,"dirt");
                }
                else{
                    setB(x,y,z,"stone");
                }
            }

            if(lake(x,z)){

                for(let y=h+1;y<=5;y++){

                    if(!getB(x,y,z)){
                        setB(x,y,z,"water");
                    }
                }
            }
        }
    }

    let tries = 0;

    while(tries < 500){

        tries++;

        const x =
            4 +
            Math.floor(Math.random() * (SIZE - 8));

        const z =
            4 +
            Math.floor(Math.random() * (SIZE - 8));

        if(lake(x,z)){
            continue;
        }

        const y = heightAt(x,z);

        if(getB(x,y,z) !== "grass"){
            continue;
        }

        if(getB(x,y+1,z)){
            continue;
        }

        const treeHeight =
            4 +
            Math.floor(Math.random() * 3);

        for(let i=1;i<=treeHeight;i++){

            if(!getB(x,y+i,z)){
                setB(x,y+i,z,"wood");
            }
        }

        const top = y + treeHeight;

        for(let ox=-2;ox<=2;ox++){
            for(let oy=-2;oy<=2;oy++){
                for(let oz=-2;oz<=2;oz++){

                    if(
                        Math.abs(ox) +
                        Math.abs(oz) +
                        Math.abs(oy) > 4
                    ){
                        continue;
                    }

                    if(Math.random() < 0.18){
                        continue;
                    }

                    const bx = x + ox;
                    const by = top + oy;
                    const bz = z + oz;

                    if(
                        inWorld(bx,by,bz) &&
                        by > BASE &&
                        !getB(bx,by,bz)
                    ){
                        setB(
                            bx,
                            by,
                            bz,
                            "leaves"
                        );
                    }
                }
            }
        }
    }
}

const player = {
    x:70.5,
    y:10,
    z:70.5,
    yaw:0,
    pitch:0,
    vy:0,
    grounded:false
};

const PW = 0.55;
const PH = 1.75;
const EYE = 1.5;

const GRAVITY = 24;
const JUMP = 8;

const WALK_SPEED = 5;
const SPRINT_SPEED = 8;

function collides(x,y,z){

    const minX = x - PW / 2;
    const maxX = x + PW / 2;

    const minY = y;
    const maxY = y + PH;

    const minZ = z - PW / 2;
    const maxZ = z + PW / 2;

    const x0 = Math.floor(minX);
    const x1 = Math.floor(maxX - 0.000001);

    const y0 = Math.floor(minY);
    const y1 = Math.floor(maxY - 0.000001);

    const z0 = Math.floor(minZ);
    const z1 = Math.floor(maxZ - 0.000001);

    for(let bx=x0;bx<=x1;bx++){
        for(let by=y0;by<=y1;by++){
            for(let bz=z0;bz<=z1;bz++){

                if(solid(bx,by,bz)){
                    return true;
                }
            }
        }
    }

    return false;
}

function moveX(amount){

    if(amount === 0) return;

    const steps =
        Math.max(
            1,
            Math.ceil(Math.abs(amount) / 0.035)
        );

    const step = amount / steps;

    for(let i=0;i<steps;i++){

        const nx = player.x + step;

        if(!collides(nx,player.y,player.z)){
            player.x = nx;
        }
        else{
            break;
        }
    }
}

function moveZ(amount){

    if(amount === 0) return;

    const steps =
        Math.max(
            1,
            Math.ceil(Math.abs(amount) / 0.035)
        );

    const step = amount / steps;

    for(let i=0;i<steps;i++){

        const nz = player.z + step;

        if(!collides(player.x,player.y,nz)){
            player.z = nz;
        }
        else{
            break;
        }
    }
}

function highestGround(){

    const minX =
        player.x - PW / 2 + 0.02;

    const maxX =
        player.x + PW / 2 - 0.02;

    const minZ =
        player.z - PW / 2 + 0.02;

    const maxZ =
        player.z + PW / 2 - 0.02;

    let best = -Infinity;

    for(
        let bx=Math.floor(minX);
        bx<=Math.floor(maxX);
        bx++
    ){

        for(
            let bz=Math.floor(minZ);
            bz<=Math.floor(maxZ);
            bz++
        ){

            for(
                let by=Math.floor(player.y)+1;
                by>=0;
                by--
            ){

                if(solid(bx,by,bz)){

                    const top = by + 1;

                    if(
                        top <= player.y + 0.12 &&
                        top > best
                    ){
                        best = top;
                    }

                    break;
                }
            }
        }
    }

    return best;
}

function moveY(amount){

    if(amount === 0) return;

    const steps =
        Math.max(
            1,
            Math.ceil(Math.abs(amount) / 0.025)
        );

    const step = amount / steps;

    for(let i=0;i<steps;i++){

        const ny = player.y + step;

        if(!collides(player.x,ny,player.z)){

            player.y = ny;
            continue;
        }

        if(step < 0){

            const ground = highestGround();

            if(ground !== -Infinity){
                player.y = ground;
            }

            player.vy = 0;
            player.grounded = true;

            return;
        }

        player.vy = 0;
        return;
    }

    const ground = highestGround();

    if(
        ground !== -Infinity &&
        Math.abs(ground - player.y) < 0.07 &&
        player.vy <= 0
    ){

        player.y = ground;
        player.vy = 0;
        player.grounded = true;

    }
    else{

        player.grounded = false;
    }
}

function spawn(){

    for(let radius=0;radius<50;radius++){

        for(let i=0;i<12;i++){

            const angle =
                i * Math.PI * 2 / 12;

            const x =
                Math.floor(
                    SIZE / 2 +
                    Math.cos(angle) * radius
                );

            const z =
                Math.floor(
                    SIZE / 2 +
                    Math.sin(angle) * radius
                );

            if(
                x < 2 ||
                x >= SIZE - 2 ||
                z < 2 ||
                z >= SIZE - 2
            ){
                continue;
            }

            const h = heightAt(x,z);

            if(getB(x,h,z) !== "grass"){
                continue;
            }

            if(getB(x,h+1,z)){
                continue;
            }

            const sy = h + 1;

            if(!collides(x+0.5,sy,z+0.5)){

                player.x = x + 0.5;
                player.y = sy;
                player.z = z + 0.5;

                player.vy = 0;
                player.grounded = true;

                return;
            }
        }
    }

    player.x = 70.5;
    player.y = BASE + 1;
    player.z = 70.5;
    player.vy = 0;
    player.grounded = true;
}

function direction(){

    const cp = Math.cos(player.pitch);
    const sp = Math.sin(player.pitch);

    const sy = Math.sin(player.yaw);
    const cy = Math.cos(player.yaw);

    return {
        x:sy * cp,
        y:-sp,
        z:cy * cp
    };
}

function raycast(){

    const d = direction();

    const sx = player.x;
    const sy = player.y + EYE;
    const sz = player.z;

    let previous = null;

    for(
        let distance=0;
        distance<=8;
        distance+=0.03
    ){

        const bx =
            Math.floor(
                sx + d.x * distance
            );

        const by =
            Math.floor(
                sy + d.y * distance
            );

        const bz =
            Math.floor(
                sz + d.z * distance
            );

        const type = getB(bx,by,bz);

        if(
            type &&
            type !== "water"
        ){

            return {
                x:bx,
                y:by,
                z:bz,
                type,
                previous
            };
        }

        previous = {
            x:bx,
            y:by,
            z:bz
        };
    }

    return null;
}

function update(dt){

    if(!started || invOpen){
        return;
    }

    let forward = 0;
    let right = 0;

    if(keys["KeyW"]) forward += 1;
    if(keys["KeyS"]) forward -= 1;

    if(keys["KeyD"]) right += 1;
    if(keys["KeyA"]) right -= 1;

    if(forward !== 0 || right !== 0){

        const length =
            Math.hypot(
                forward,
                right
            );

        forward /= length;
        right /= length;

        const speed =
            (
                keys["ShiftLeft"] ||
                keys["ShiftRight"]
            )
                ? SPRINT_SPEED
                : WALK_SPEED;

        const sin =
            Math.sin(player.yaw);

        const cos =
            Math.cos(player.yaw);

        const forwardX = sin;
        const forwardZ = cos;

        const rightX = cos;
        const rightZ = -sin;

        const dx =
            (
                forwardX * forward +
                rightX * right
            ) *
            speed *
            dt;

        const dz =
            (
                forwardZ * forward +
                rightZ * right
            ) *
            speed *
            dt;

        moveX(dx);
        moveZ(dz);
    }

    player.vy -= GRAVITY * dt;

    moveY(
        player.vy * dt
    );

    player.x =
        Math.max(
            1,
            Math.min(
                SIZE - 1,
                player.x
            )
        );

    player.z =
        Math.max(
            1,
            Math.min(
                SIZE - 1,
                player.z
            )
        );

    if(player.y < BASE){
        spawn();
    }
}

function project(x,y,z){

    const dx = x - player.x;
    const dy = y - (player.y + EYE);
    const dz = z - player.z;

    const s =
        Math.sin(player.yaw);

    const c =
        Math.cos(player.yaw);

    let X =
        dx * c -
        dz * s;

    let Z =
        dx * s +
        dz * c;

    const cp =
        Math.cos(player.pitch);

    const sp =
        Math.sin(player.pitch);

    let Y =
        dy * cp -
        Z * sp;

    Z =
        dy * sp +
        Z * cp;

    if(Z <= 0.12){
        return null;
    }

    const focal = 540;

    return {
        x:W / 2 + X * focal / Z,
        y:H / 2 - Y * focal / Z,
        z:Z
    };
}

function shade(hex,factor){

    const r =
        parseInt(
            hex.slice(1,3),
            16
        );

    const g =
        parseInt(
            hex.slice(3,5),
            16
        );

    const b =
        parseInt(
            hex.slice(5,7),
            16
        );

    return (
        "rgb(" +
        Math.floor(r * factor) +
        "," +
        Math.floor(g * factor) +
        "," +
        Math.floor(b * factor) +
        ")"
    );
}

function brightness(n){

    if(n[1] > 0) return 1.12;
    if(n[1] < 0) return 0.65;
    if(n[0] < 0) return 0.82;
    if(n[0] > 0) return 0.95;
    if(n[2] > 0) return 1;
    return 0.76;
}

function drawBlock(x,y,z,type){

    const block = BLOCKS[type];

    for(const face of faces){

        const nx = x + face.n[0];
        const ny = y + face.n[1];
        const nz = z + face.n[2];

        if(solid(nx,ny,nz)){
            continue;
        }

        const points = [];

        for(const vertex of face.v){

            const point =
                project(
                    x + vertex[0],
                    y + vertex[1],
                    z + vertex[2]
                );

            if(!point){

                points.length = 0;
                break;
            }

            points.push(point);
        }

        if(points.length !== 4){
            continue;
        }

        ctx.beginPath();

        ctx.moveTo(
            points[0].x,
            points[0].y
        );

        for(let i=1;i<4;i++){

            ctx.lineTo(
                points[i].x,
                points[i].y
            );
        }

        ctx.closePath();

        ctx.fillStyle =
            shade(
                block.c,
                brightness(face.n)
            );

        ctx.fill();

        ctx.strokeStyle =
            "rgba(0,0,0,.12)";

        ctx.stroke();
    }
}

function render(){

    const sky =
        ctx.createLinearGradient(
            0,
            0,
            0,
            H
        );

    sky.addColorStop(0,"#61b3ea");
    sky.addColorStop(1,"#daf4ff");

    ctx.fillStyle = sky;

    ctx.fillRect(
        0,
        0,
        W,
        H
    );

    if(!started){
        return;
    }

    const px =
        Math.floor(player.x);

    const pz =
        Math.floor(player.z);

    const list = [];

    for(
        let x=Math.max(0,px-VIEW);
        x<=Math.min(SIZE-1,px+VIEW);
        x++
    ){

        for(
            let z=Math.max(0,pz-VIEW);
            z<=Math.min(SIZE-1,pz+VIEW);
            z++
        ){

            const distance2 =
                (x-player.x) ** 2 +
                (z-player.z) ** 2;

            if(
                distance2 >
                VIEW * VIEW
            ){
                continue;
            }

            for(
                let y=BASE;
                y<HEIGHT;
                y++
            ){

                const type =
                    getB(x,y,z);

                if(!type){
                    continue;
                }

                let visible = false;

                for(const face of faces){

                    if(
                        !solid(
                            x + face.n[0],
                            y + face.n[1],
                            z + face.n[2]
                        )
                    ){
                        visible = true;
                        break;
                    }
                }

                if(!visible){
                    continue;
                }

                list.push({
                    x,
                    y,
                    z,
                    type,
                    d:
                        distance2 +
                        (y-player.y) ** 2 * 0.25
                });
            }
        }
    }

    list.sort(
        (a,b)=>b.d-a.d
    );

    for(const block of list){

        drawBlock(
            block.x,
            block.y,
            block.z,
            block.type
        );
    }
}
